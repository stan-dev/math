66 // odeint controls the max of the weighted errors, not their RMS
67 static realtype max_norm(N_Vector x, N_Vector w) {
68 Eigen::Map<const Eigen::ArrayXd> x_map(NV_DATA_S(x), NV_LENGTH_S(x));
69 Eigen::Map<const Eigen::ArrayXd> w_map(NV_DATA_S(w), NV_LENGTH_S(w));
70 return (x_map * w_map).abs().maxCoeff();
71 }

158 std::vector<Eigen::Matrix<T_Return, Eigen::Dynamic, 1>> operator()() {
159 std::vector<Eigen::Matrix<T_Return, Eigen::Dynamic, 1>> y;
160 y.reserve(ts_.size());
161
162 auto free_mem = [](void* mem) { ERKStepFree(&mem); };
163 std::unique_ptr<void, decltype(free_mem)> mem_ptr(
164 ERKStepCreate(&arkode_integrator::erk_rhs, value_of(t0_), nv_state_,
165 sundials_context_),
166 free_mem);
167 void* arkode_mem = mem_ptr.get();
168 if (arkode_mem == nullptr) {
169 throw std::runtime_error("ERKStepCreate failed to allocate memory");
170 }
171
172 CHECK_ARKODE_CALL(
173 ERKStepSetUserData(arkode_mem, reinterpret_cast<void*>(this)));
174 CHECK_ARKODE_CALL(ERKStepSetTableNum(arkode_mem, Table));
175 CHECK_ARKODE_CALL(ERKStepSStolerances(arkode_mem, relative_tolerance_,
176 absolute_tolerance_));
177 CHECK_ARKODE_CALL(ERKStepSetMaxNumSteps(arkode_mem, max_num_steps_));
178
179 // odeint: h *= 0.9 * err^(-1/5) limited to [0.2, 4.5], h kept if err >= 0.5
180 realtype adapt_params[3] = {1.0, 0.0, 0.0};
181 CHECK_ARKODE_CALL(ERKStepSetInitStep(arkode_mem, 0.1));
182 CHECK_ARKODE_CALL(ERKStepSetAdaptivityMethod(arkode_mem, ARK_ADAPT_I, 0, 1,
183 adapt_params));
184 CHECK_ARKODE_CALL(ERKStepSetSafetyFactor(arkode_mem, 0.9));
185 CHECK_ARKODE_CALL(ERKStepSetErrorBias(arkode_mem, 1.0));
186 CHECK_ARKODE_CALL(ERKStepSetMaxGrowth(arkode_mem, 4.5));
187 CHECK_ARKODE_CALL(ERKStepSetMaxFirstGrowth(arkode_mem, 4.5));
188 CHECK_ARKODE_CALL(ERKStepSetMinReduction(arkode_mem, 0.2));
189 CHECK_ARKODE_CALL(
190 ERKStepSetFixedStepBounds(arkode_mem, 0.9, 0.9 * std::pow(0.5, -0.2)));
191 if (!DenseOutput) {
192 // never evaluated, but avoids the Hermite interpolant's extra RHS calls
193 CHECK_ARKODE_CALL(
194 ERKStepSetInterpolantType(arkode_mem, ARK_INTERP_LAGRANGE));
195 }
196
197 double t_init = value_of(t0_);
198 for (size_t n = 0; n < ts_.size(); ++n) {
199 double t_final = value_of(ts_[n]);
200
201 if (t_final != t_init) {
202 if (!DenseOutput) {
203 CHECK_ARKODE_CALL(ERKStepSetStopTime(arkode_mem, t_final));
204 }
205 int flag = ERKStepEvolve(arkode_mem, t_final, nv_state_, &t_init,
206 ARK_NORMAL);
207 if (flag == ARK_TOO_MUCH_WORK) {
208 throw_domain_error(function_name_, "", t_final,
209 "Failed to integrate to next output time (",
210 ") in less than max_num_steps steps");
211 }
212 arkode_check(flag, "ERKStepEvolve");
213 }
214
215 y.emplace_back(math::apply(
216 [&](const auto&... args_ref) {
217 return ode_store_sensitivities(f_, coupled_state_, y0_, t0_, ts_[n],
218 msgs_, args_ref...);
219 },
220 args_tuple_));
221
222 t_init = t_final;
223 }
224
225 return y;
226 }
