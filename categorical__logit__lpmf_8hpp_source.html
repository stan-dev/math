35template <bool propto, typename T_prob, require_col_vector_t<T_prob>* = nullptr>
36inline return_type_t<T_prob> categorical_logit_lpmf(int n, const T_prob& beta) {
37 static constexpr const char* function = "categorical_logit_lpmf";
38 check_bounded(function, "categorical outcome out of support", n, 1,
39 beta.size());
40 ref_type_t<T_prob> beta_ref = beta;
41
42 // Autodiff args must be finite data may be +/-inf
43 if constexpr (is_constant_v<T_prob>) {
44 // Data Case: Throws in nan and all -inf case
45 check_not_nan(function, "log odds parameter", beta_ref);
46 check_greater(function, "log odds parameter", beta_ref.maxCoeff(),
47 NEGATIVE_INFTY);
48 } else {
49 // Autodiff Case: Throws in non-finite case
50 check_finite(function, "log odds parameter", beta_ref);
51 }
52
53 if constexpr (!include_summand<propto, T_prob>::value) {
54 return 0.0;
55 }
56
57 // softmax is NaN with +inf, so split probability evenly over the
58 // +inf entries and give every other entry zero probability
59 if constexpr (is_constant_v<T_prob>) {
60 int num_infty = (beta_ref.array() == INFTY).count();
61 if (num_infty > 0) {
62 return beta_ref.coeff(n - 1) == INFTY ? -std::log(num_infty)
63 : NEGATIVE_INFTY;
64 }
65 }
66
67 // FIXME: wasteful vs. creating term (n-1) if not vectorized
68 return beta_ref.coeff(n - 1)
69 - log_sum_exp(beta_ref); // == log_softmax(beta)(n-1);
70}
