40 // Throws in nan and all -inf case

43 check_greater(function, "Log odds parameter", beta.maxCoeff(),

46 auto is_pos_inf = (beta.array() == INFTY);
47 int num_infty = is_pos_inf.count();
48
49 Eigen::VectorXd theta;
50
51 // softmax is NaN with +inf, so split probability evenly over the
52 // +inf entries and give every other entry zero probability
53 if (num_infty > 0) {
54 theta = is_pos_inf.template cast<double>() / num_infty;
55 } else {
56 theta = softmax(beta);
57 }
58
59 variate_generator<RNG&, uniform_01<> > uniform01_rng(rng, uniform_01<>());
60 Eigen::VectorXd index = cumulative_sum(theta);
61
62 // index may sum to just below 1, so stop at the last nonzero entry
63 int last = theta.size() - 1;
64 while (theta(last) == 0) {
65 last--;
66 }
67 double c = uniform01_rng();

71 }
