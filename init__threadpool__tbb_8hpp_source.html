46inline int get_num_threads() {

49 if (env_stan_num_threads == nullptr) {
50 return 1;
51 }
52
53 const std::string_view value(env_stan_num_threads);
54 int num_threads;
55 const auto [end, error]
56 = std::from_chars(value.begin(), value.end(), num_threads);
