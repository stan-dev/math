37template <CorrOrder Order>
38inline Eigen::Index cholesky_corr_position(Eigen::Index row, Eigen::Index col,
39 Eigen::Index K) {
40 if constexpr (Order == CorrOrder::column) {
41 return col * (K - 1) - (col * (col - 1)) / 2 + (row - col - 1);
42 } else {
43 return (row * (row - 1)) / 2 + col;
44 }
45}

60inline auto cholesky_corr_log_jacobian_coefficients(Eigen::Index K) {
61 const Eigen::Index k_choose_2 = (K * (K - 1)) / 2;
62 return make_holder(
63 [](auto&& pos) {
64 const auto row = (0.5 * (1.0 + (1.0 + 8.0 * pos).sqrt())).floor();
65 return (row + 1.0 - (pos - 0.5 * row * (row - 1.0))).matrix();

114template <CorrOrder Order, typename EigVec, typename LogCoshVec,
115 require_all_eigen_vector_t<EigVec, LogCoshVec>* = nullptr>
116inline Eigen::Matrix<value_type_t<EigVec>, Eigen::Dynamic, Eigen::Dynamic>
117cholesky_corr_factor(EigVec&& y, LogCoshVec&& log_cosh_y, Eigen::Index K) {
118 using T_scalar = value_type_t<EigVec>;
119 Eigen::Matrix<T_scalar, Eigen::Dynamic, Eigen::Dynamic> L
120 = Eigen::Matrix<T_scalar, Eigen::Dynamic, Eigen::Dynamic>::Zero(K, K);
121 if (K == 0) {
122 return L;
123 }
124 decltype(auto) y_ref = to_ref(std::forward<EigVec>(y));
125 decltype(auto) log_cosh_y_ref = to_ref(std::forward<LogCoshVec>(log_cosh_y));
126 L.coeffRef(0, 0) = 1;
127 for (Eigen::Index row = 1; row < K; ++row) {
128 T_scalar acc = 0;
129 for (Eigen::Index col = 0; col < row; ++col) {
130 const Eigen::Index pos = cholesky_corr_position<Order>(row, col, K);
131 L.coeffRef(row, col) = tanh(y_ref.coeff(pos)) * exp(-acc);
132 acc += log_cosh_y_ref.coeff(pos);
133 }
134 L.coeffRef(row, row) = exp(-acc);
135 }
136 return L;
137}

177cholesky_corr_constrain(EigVec&& y, int K, Lp& lp) {
178 int k_choose_2 = (K * (K - 1)) / 2;
