63template <typename T, require_eigen_col_vector_t<T>* = nullptr>
64inline Eigen::Matrix<value_type_t<T>, Eigen::Dynamic, Eigen::Dynamic>
65cov_matrix_constrain_lkj(T&& x, size_t k, return_type_t<T>& lp) {
66 size_t k_choose_2 = (k * (k - 1)) / 2;
67 decltype(auto) x_ref = to_ref(std::forward<T>(x));
68 check_size_match("cov_matrix_constrain_lkj", "x.size()", x_ref.size(),
69 "k_choose_2 + k", k_choose_2 + k);
70 auto cpcs = x_ref.head(k_choose_2);
71 auto log_sds = x_ref.tail(k);
72 decltype(auto) log_cosh_cpcs = to_ref(log_cosh(cpcs));
73 lp -= dot_product(internal::corr_matrix_log_jacobian_coefficients(k),
74 log_cosh_cpcs);
75 lp += (k + 1.0) * sum(log_sds) + k * LOG_TWO;
76 return multiply_lower_tri_self_transpose(diag_pre_multiply(
77 to_ref(positive_constrain(log_sds)),
78 internal::cholesky_corr_factor<internal::CorrOrder::column>(
79 cpcs, log_cosh_cpcs, k)));
80}
