40template <typename T, typename M, typename S,

43inline auto offset_multiplier_constrain(const T& x, const M& mu,
44 const S& sigma) {
45 const auto& mu_ref = to_ref(mu);
46 const auto& sigma_ref = to_ref(sigma);
47 if constexpr (is_matrix<T>::value && is_matrix<M>::value) {
48 check_matching_dims("offset_multiplier_constrain", "x", x, "mu", mu);
49 }
50 if constexpr (is_matrix<T>::value && is_matrix<S>::value) {
51 check_matching_dims("offset_multiplier_constrain", "x", x, "sigma", sigma);
52 } else if constexpr (is_matrix<M>::value && is_matrix<S>::value) {
53 check_matching_dims("offset_multiplier_constrain", "mu", mu, "sigma",
54 sigma);
55 }
56
57 check_finite("offset_multiplier_constrain", "offset", value_of_rec(mu_ref));
58 check_positive_finite("offset_multiplier_constrain", "multiplier",
59 value_of_rec(sigma_ref));
60 return stan::math::eval(fma(sigma_ref, x, mu_ref));
61}

91 require_convertible_t<return_type_t<T, M, S>, Lp>* = nullptr,
92 require_all_not_nonscalar_prim_or_rev_kernel_expression_t<
93 T, M, S>* = nullptr>
94inline auto offset_multiplier_constrain(const T& x, const M& mu, const S& sigma,
95 Lp& lp) {
96 const auto& mu_ref = to_ref(mu);
97 const auto& sigma_ref = to_ref(sigma);
98 if constexpr (is_matrix<T>::value && is_matrix<M>::value) {
99 check_matching_dims("offset_multiplier_constrain", "x", x, "mu", mu);
100 }
101 if constexpr (is_matrix<T>::value && is_matrix<S>::value) {
102 check_matching_dims("offset_multiplier_constrain", "x", x, "sigma", sigma);
103 } else if constexpr (is_matrix<M>::value && is_matrix<S>::value) {
104 check_matching_dims("offset_multiplier_constrain", "mu", mu, "sigma",
105 sigma);
106 }
107
108 check_finite("offset_multiplier_constrain", "offset", value_of_rec(mu_ref));
109 check_positive_finite("offset_multiplier_constrain", "multiplier",
110 value_of_rec(sigma_ref));
111 if (math::size(sigma_ref) == 1) {
112 lp += sum(lmultiply(math::size(x), sigma_ref));
113 } else {
114 lp += sum(log(sigma_ref));
115 }
116 return stan::math::eval(fma(sigma_ref, x, mu_ref));
117}
