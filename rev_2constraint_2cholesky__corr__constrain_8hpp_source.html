32template <CorrOrder Order, typename T, require_var_vector_t<T>* = nullptr>
33inline var_value<Eigen::MatrixXd> cholesky_corr_factor(T&& y, Eigen::Index K) {
34 using Eigen::Index;
35 if (K == 0) {
36 return Eigen::MatrixXd();
37 }
38 auto arena_y = to_arena(std::forward<T>(y));
39 arena_t<Eigen::VectorXd> tanh_y(arena_y.size());
40 // derivative of each element with respect to its own unconstrained value
41 arena_t<Eigen::VectorXd> d_own(arena_y.size());
42 arena_t<Eigen::MatrixXd> L_val = Eigen::MatrixXd::Zero(K, K);
43 L_val.coeffRef(0, 0) = 1;
44 for (Index row = 1; row < K; ++row) {
45 // exp(-acc), as the product of sech(y) over the earlier elements of the
46 // row
47 double exp_neg_acc = 1.0;
48 for (Index col = 0; col < row; ++col) {
49 const Index pos = cholesky_corr_position<Order>(row, col, K);
50 const double y_val = arena_y.val().coeff(pos);
51 const double sech_val = sech(y_val);
52 tanh_y.coeffRef(pos) = tanh(y_val);
53 d_own.coeffRef(pos) = sech_val * sech_val * exp_neg_acc;
54 L_val.coeffRef(row, col) = tanh_y.coeff(pos) * exp_neg_acc;
55 exp_neg_acc *= sech_val;
56 }
57 L_val.coeffRef(row, row) = exp_neg_acc;
58 }
59 var_value<Eigen::MatrixXd> L = L_val;
60 reverse_pass_callback([arena_y, L, tanh_y, d_own, K]() mutable {
61 for (Index row = 1; row < K; ++row) {
62 // sum of adj * val over the elements of the row after column col, which
63 // all depend on y_col through the shared exp(-acc)
64 double running_sum = L.adj().coeff(row, row) * L.val().coeff(row, row);
65 for (Index col = row - 1; col >= 0; --col) {
66 const Index pos = cholesky_corr_position<Order>(row, col, K);
67 const double adj_rc = L.adj().coeff(row, col);
68 arena_y.adj().coeffRef(pos)
69 += adj_rc * d_own.coeff(pos) - tanh_y.coeff(pos) * running_sum;
70 running_sum += adj_rc * L.val().coeff(row, col);
71 }
72 }
73 });
74 return L;
75}

90inline var_value<Eigen::MatrixXd> cholesky_corr_constrain(T&& y, int K) {
91 int k_choose_2 = (K * (K - 1)) / 2;
92 check_size_match("cholesky_corr_constrain", "y.size()", y.size(),
