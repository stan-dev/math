40 using cdouble = std::complex<double>;
41 using ret_type

54 auto arena_A = to_arena(std::forward<MatA>(A));
55 auto arena_B = to_arena(std::forward<MatB>(B));
56 auto arena_C = to_arena(std::forward<MatC>(C));

58 const Eigen::Index m = B.rows();

60 // ---- forward pass: identical to prim, on values ----
61 Eigen::ComplexSchur<plain_type_t<decltype(value_of(arena_A))>> schur_A(
62 value_of(arena_A));
63 auto arena_U = to_arena(schur_A.matrixU());
64 auto arena_S = to_arena(schur_A.matrixT());
65
66 Eigen::ComplexSchur<plain_type_t<decltype(value_of(arena_B))>> schur_B(
67 value_of(arena_B));

69 auto arena_T = to_arena(schur_B.matrixT());
70
71 // Y holds D = U^H C V initially, overwritten column by column
72 auto Y = (arena_U.adjoint() * value_of(arena_C) * arena_V).eval();
73 plain_type_t<decltype(arena_S)> Sshift = arena_S;
74 for (Eigen::Index i = 0; i < m; ++i) {
75 Y.col(i).noalias() -= Y.leftCols(i) * arena_T.col(i).head(i);
76 Sshift.diagonal() = arena_S.diagonal().array() + arena_T(i, i);
77 Sshift.template triangularView<Eigen::Upper>().solveInPlace(Y.col(i));
78 }

83 // ---- reverse pass ----
84 reverse_pass_callback([arena_A, arena_B, arena_C, arena_U, arena_S, arena_V,
85 arena_T, arena_X, res]() mutable {

87 const Eigen::Index cols = arena_T.rows();

89 // G = adj(Re X) + i * adj(Im X)

91 for (Eigen::Index c = 0; c < cols; ++c) {

94 }
95 }

97 // W holds E = U^H G V initially; solve S^H W + W T^H = E,
98 // overwritten column by column from last to first
99 auto W = (arena_U.adjoint() * G * arena_V).eval();
100 auto Sshift_r = arena_S.eval();
101 for (Eigen::Index j = cols - 1; j >= 0; --j) {
102 const Eigen::Index k = cols - 1 - j; // already-solved columns
103 W.col(j).noalias() -= W.rightCols(k) * arena_T.row(j).tail(k).adjoint();
104 Sshift_r.diagonal() = arena_S.diagonal().array() + arena_T(j, j);
105 Sshift_r.template triangularView<Eigen::Upper>().adjoint().solveInPlace(
106 W.col(j));
107 }
108 const auto Z = (arena_U * W * arena_V.adjoint()).eval();
109
110 // Re/Im of the gradient go to the adjoints of Re/Im of the input;
111 // real inputs only receive Re(grad)
112 auto accumulate = [](auto& in, const Eigen::MatrixXcd& g) {
113 using in_t = std::decay_t<decltype(in)>;
114 if constexpr (is_complex<value_type_t<in_t>>::value) {
115 const Eigen::Index rows = g.rows();
116 const Eigen::Index cols = g.cols();
117 for (Eigen::Index c = 0; c < cols; ++c) {
118 for (Eigen::Index r = 0; r < rows; ++r) {
119 in(r, c).real().adj() += g(r, c).real();
120 in(r, c).imag().adj() += g(r, c).imag();
121 }
122 }
123 } else {
124 in.adj() += g.real();
125 }
126 };
127
128 if constexpr (is_var<base_type_t<MatA>>::value) {
129 accumulate(arena_A, -(Z * arena_X.adjoint()));
130 }
131 if constexpr (is_var<base_type_t<MatB>>::value) {
132 accumulate(arena_B, -(arena_X.adjoint() * Z));
133 }
134 if constexpr (is_var<base_type_t<MatC>>::value) {
135 accumulate(arena_C, Z);
136 }
137 });
138 return ret_type(res);
139}
