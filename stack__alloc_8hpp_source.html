59STAN_STRONG_INLINE char* align_up(char* ptr, size_t alignment) {
60 const uintptr_t address = reinterpret_cast<uintptr_t>(ptr);
61 const uintptr_t aligned
62 = (address + alignment - 1) & ~static_cast<uintptr_t>(alignment - 1);
63 return ptr + (aligned - address);
64}

92 // block
93 // next three for keeping track of nested allocations on top of stack:
94 std::vector<size_t> nested_cur_blocks_;
95 std::vector<char*> nested_next_locs_;

107 STAN_STRONG_INLINE char* move_to_next_block(size_t len, size_t alignment) {
108 // Blocks are 8 byte aligned, so leave room to align the start.
109 const size_t needed = alignment > 8 ? len + alignment - 8 : len;
110 ++cur_block_;
111 // Find the next block (if any) containing at least needed bytes.
112 while ((cur_block_ < blocks_.size()) && (sizes_[cur_block_] < needed)) {
113 ++cur_block_;
114 }
115 // Allocate a new block if necessary.
116 if (unlikely(cur_block_ >= blocks_.size())) {
117 // New block should be max(2*size of last block, needed) bytes.
118 size_t newsize = sizes_.back() * 2;
119 if (newsize < needed) {
120 newsize = needed;
121 }
122 blocks_.push_back(internal::eight_byte_aligned_malloc(newsize));
123 if (!blocks_.back()) {
124 throw std::bad_alloc();
125 }
126 sizes_.push_back(newsize);
127 }
128 char* result = internal::align_up(blocks_[cur_block_], alignment);
129 // Get the object's state back in order.
130 next_loc_ = result + len;
131 cur_block_end_ = blocks_[cur_block_] + sizes_[cur_block_];
132 return result;
133 }
134
135 public:
145 explicit stack_alloc(size_t initial_nbytes = internal::DEFAULT_INITIAL_NBYTES)
146 : blocks_(1, internal::eight_byte_aligned_malloc(initial_nbytes)),
147 sizes_(1, initial_nbytes),
148 cur_block_(0),
149 cur_block_end_(blocks_[0] + initial_nbytes),
150 next_loc_(blocks_[0]) {
151 if (!blocks_[0]) {
152 throw std::bad_alloc(); // no msg allowed in bad_alloc ctor
153 }
154 }

186 STAN_STRONG_INLINE void* alloc(size_t len, size_t alignment = 8) {
187 size_t pad = len % 8 == 0 ? 0 : 8 - len % 8;
188
189 // Typically, just return and increment the next location.
190 char* result
191 = alignment <= 8 ? next_loc_ : internal::align_up(next_loc_, alignment);
192 next_loc_ = result + len + pad;
193 // Occasionally, we have to switch blocks.
194 if (unlikely(next_loc_ >= cur_block_end_)) {
195 result = move_to_next_block(len + pad, alignment);
196 }
197 return reinterpret_cast<void*>(result);
198 }
