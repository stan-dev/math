# A single set of flags and dependencies keeps PCH producers and consumers
# compatible. Do not enable IPO or add target-specific optimization flags.
# Stan's vendored gtest_main initializes the MPI cluster and includes Stan
# headers even when MPI is disabled.
target_link_libraries(gtest_main PRIVATE stanmath)
if(STAN_MPI)
  target_link_libraries(gtest_main PRIVATE stan_mpi)
endif()
add_library(stan_test_options INTERFACE)
target_link_libraries(stan_test_options INTERFACE stanmath gtest_main)
if(STAN_MPI)
  target_link_libraries(stan_test_options INTERFACE stan_mpi)
endif()
if(MSVC)
  target_compile_options(stan_test_options INTERFACE ${STAN_TEST_OPTIMIZATION_FLAGS} /W3)
else()
  target_compile_options(stan_test_options INTERFACE
    ${STAN_TEST_OPTIMIZATION_FLAGS} -Wall -Wno-deprecated-declarations -Wno-sign-compare)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(stan_test_options INTERFACE
      -Werror=invalid-pch -Wno-misleading-indentation)
  endif()
endif()

function(stan_add_pch target)
  if(STAN_TEST_PCH)
    add_library(${target} OBJECT EXCLUDE_FROM_ALL "${PROJECT_SOURCE_DIR}/cmake/pch.cpp")
    set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE FALSE)
    target_link_libraries(${target} PRIVATE stan_test_options)
    target_precompile_headers(${target} PRIVATE ${ARGN})
  endif()
endfunction()

function(stan_configure_test target pch)
  set_target_properties(${target} PROPERTIES POSITION_INDEPENDENT_CODE FALSE)
  target_link_libraries(${target} PRIVATE stan_test_options)
  if(STAN_TEST_PCH AND pch)
    if(TARGET ${pch})
      target_precompile_headers(${target} REUSE_FROM ${pch})
    endif()
  endif()
endfunction()

function(stan_add_test_object target source pch)
  add_library(${target} OBJECT EXCLUDE_FROM_ALL "${source}")
  stan_configure_test(${target} "${pch}")
  get_source_file_property(generator_target "${source}" STAN_GENERATOR_TARGET)
  if(generator_target)
    # Every consuming executable needs generation before object compilation.
    add_dependencies(${target} ${generator_target})
  endif()
endfunction()

function(stan_add_test_binary target)
  # Xcode needs a real source alongside object-library references.
  add_executable(${target} "${PROJECT_SOURCE_DIR}/cmake/pch.cpp" ${ARGN})
  stan_configure_test(${target} "")
  set_property(TARGET ${target} PROPERTY STAN_TEST_OBJECTS "${ARGN}")
endfunction()

# Directory and per-file executables share objects with per-source PCH choices.
function(add_gtest_grouped_test test_directory folder_test_name target_pch)
  set(sources ${ARGN})
  if(NOT sources)
    file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${test_directory}/*test.cpp")
  endif()
  set(directory_keys "")
  foreach(src IN LISTS sources)
    file(RELATIVE_PATH relative_source "${test_directory}" "${src}")
    set(repo_relative "test/${folder_test_name}/${relative_source}")
    if(folder_test_name STREQUAL "unit_math")
      set(repo_relative "test/unit/math/${relative_source}")
    elseif(folder_test_name STREQUAL "prob_math")
      set(repo_relative "test/prob/${relative_source}")
    elseif(folder_test_name STREQUAL "expressions")
      set(repo_relative "test/expressions/${relative_source}")
    endif()
    if(STAN_TEST_FILTER AND NOT repo_relative MATCHES "${STAN_TEST_FILTER}")
      continue()
    endif()
    set(matches_name_filters TRUE)
    foreach(filter IN LISTS STAN_TEST_NAME_FILTERS)
      string(FIND "${repo_relative}" "${filter}" match_index)
      if(match_index EQUAL -1)
        set(matches_name_filters FALSE)
      endif()
    endforeach()
    if(NOT matches_name_filters)
      continue()
    endif()
    if(NOT STAN_OPENCL AND repo_relative MATCHES "/opencl/")
      continue()
    endif()
    if(NOT STAN_MPI AND repo_relative MATCHES "/[^/]*mpi[^/]*test\\.cpp$")
      continue()
    endif()
    string(REPLACE "/" "_" name "${repo_relative}")
    string(REGEX REPLACE "\\.cpp$" "" name "${name}")
    set(pch "${target_pch}")
    if(folder_test_name STREQUAL "unit_math")
      string(REGEX MATCH "^[^/]+" layer "${relative_source}")
      if(TARGET unit_${layer}_pch)
        set(pch unit_${layer}_pch)
      endif()
      # Precompile the AD helpers only for sources that already include them;
      # their global aliases must not leak into unrelated tests.
      file(STRINGS "${src}" ad_include REGEX "^#include <test/unit/math/test_ad\\.hpp>")
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${src}")
      if(ad_include AND TARGET unit_ad_pch)
        set(pch unit_ad_pch)
      endif()
      file(STRINGS "${src}" configuration_macros
        REGEX "^#[ \t]*(define|undef)[ \t]+(STAN_|EIGEN_)")
      if(configuration_macros)
        # Such a source must set its macros before including Stan/Eigen.
        set_source_files_properties("${src}" PROPERTIES SKIP_PRECOMPILE_HEADERS ON)
        set(pch "")
      endif()
    endif()
    stan_add_test_object(${name}_objects "${src}" "${pch}")
    set(object "$<TARGET_OBJECTS:${name}_objects>")
    stan_add_test_binary(${name} "${object}")
    if(STAN_MPI)
      set(mpi_ranks 1)
      if(repo_relative MATCHES "mpi_")
        set(mpi_ranks 2)
      endif()
      add_test(NAME ${name} COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${mpi_ranks}
        ${MPIEXEC_PREFLAGS} $<TARGET_FILE:${name}> ${MPIEXEC_POSTFLAGS})
    else()
      add_test(NAME ${name} COMMAND ${name})
    endif()
    get_filename_component(relative_dir "${relative_source}" DIRECTORY)
    string(REPLACE "/" ";" labels "${relative_dir}")
    set_tests_properties(${name} PROPERTIES
      WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}"
      LABELS "${folder_test_name};${labels}")

    set(key "${relative_dir}")
    while(TRUE)
      if(key STREQUAL "")
        set(key ROOT)
      endif()
      string(REPLACE "/" "_" key "${key}")
      list(APPEND directory_keys "${key}")
      list(APPEND subtests_${key} "${name}")
      list(APPEND objects_${key} "${object}")
      if(key STREQUAL "ROOT")
        break()
      endif()
      # The next ancestor comes from the path, not the underscore target name.
      get_filename_component(relative_dir "${relative_dir}" DIRECTORY)
      set(key "${relative_dir}")
    endwhile()
  endforeach()
  list(REMOVE_DUPLICATES directory_keys)
  foreach(key IN LISTS directory_keys)
    if(key STREQUAL "ROOT")
      set(prefix "${folder_test_name}")
    else()
      set(prefix "${folder_test_name}_${key}")
    endif()
    add_custom_target(${prefix}_subtests DEPENDS ${subtests_${key}})
    stan_add_test_binary(${prefix}_all_tests ${objects_${key}})
    set_target_properties(${prefix}_all_tests PROPERTIES EXCLUDE_FROM_ALL TRUE)
  endforeach()
endfunction()

if(STAN_BUILD_EXPRESSION_TESTS)
  set(STAN_EXPRESSION_TEST_DIR "${PROJECT_BINARY_DIR}/expression-sources"
    CACHE PATH "Directory containing generated expression tests")
  file(GLOB expression_sources CONFIGURE_DEPENDS "${STAN_EXPRESSION_TEST_DIR}/*test.cpp")
  if(NOT expression_sources)
    message(FATAL_ERROR "Generate expression tests before configuring STAN_BUILD_EXPRESSION_TESTS")
  endif()
  # The expression helper has no include guard; precompile its heavy guarded
  # dependencies so each generated source can still include the helper once.
  stan_add_pch(expression_pch <stan/math/mix.hpp> <gtest/gtest.h>)
  add_gtest_grouped_test("${STAN_EXPRESSION_TEST_DIR}" expressions expression_pch ${expression_sources})
endif()

if(STAN_TEST_HEADERS)
  # A PCH would conceal missing includes. Force each header into an otherwise
  # empty translation unit, and do not inherit the tests' precompiled headers.
  file(GLOB_RECURSE headers CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/stan/*.hpp")
  add_custom_target(test-headers)
  foreach(header IN LISTS headers)
    file(RELATIVE_PATH relative_header "${PROJECT_SOURCE_DIR}" "${header}")
    if(NOT STAN_OPENCL AND relative_header MATCHES "/opencl/")
      continue()
    endif()
    if(NOT STAN_MPI AND relative_header MATCHES "/mpi_[^/]+\\.hpp$")
      continue()
    endif()
    string(REPLACE "/" "_" name "${relative_header}")
    string(REGEX REPLACE "\\.hpp$" "-test" name "${name}")
    add_library(${name} OBJECT EXCLUDE_FROM_ALL "${PROJECT_SOURCE_DIR}/cmake/pch.cpp")
    target_link_libraries(${name} PRIVATE stanmath)
    if(MSVC)
      target_compile_options(${name} PRIVATE "/FI${header}")
    else()
      target_compile_options(${name} PRIVATE -O0 -include "${header}" -Wunused-local-typedefs)
    endif()
    add_dependencies(test-headers ${name})
  endforeach()
endif()
